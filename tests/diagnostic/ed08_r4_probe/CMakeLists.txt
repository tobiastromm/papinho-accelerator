cmake_minimum_required(VERSION 3.16)
project(ed08_r4_integrated C)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
string(REGEX REPLACE "/W[0-4]" "" CMAKE_C_FLAGS "${CMAKE_C_FLAGS}")
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "MSVC x64 required")
endif()
set(tid win32-x64-msvc-19.51-openssl3)
set(normal "${STAGING}/sdk-normal")
set(r4 "${STAGING}/sdk-r4")
file(SHA256 "${r4}/lib/papinho_secure_transport_API3_SPI4_TEST_ONLY_R4.lib" identity)
if(NOT identity STREQUAL "1450c01c2c1132566cdac2a64be8e84b72478800ba164207b7c7f592cae9ded5")
 message(FATAL_ERROR "R4 library mismatch")
endif()
add_executable(ed08_r4_probe ed08_r4_probe.c consumer/src/runtime/connection.c consumer/src/transports/transport_connection.c consumer/src/network/network_endpoint.c consumer/src/network/ip_address.c consumer/src/security/pst_secure_processor.c consumer/src/security/pst_secure_scheduler.c consumer/src/security/pst_secure_transport_adapter.c)
target_include_directories(ed08_r4_probe PRIVATE "${normal}/include" "${r4}/private" consumer/include consumer/src/runtime consumer/src/transports consumer/src/network consumer/src/security)
target_compile_options(ed08_r4_probe PRIVATE /W4 /WX)
target_link_libraries(ed08_r4_probe PRIVATE "${r4}/lib/papinho_secure_transport_API3_SPI4_TEST_ONLY_R4.lib" "${normal}/lib/${tid}/libssl.lib" "${normal}/lib/${tid}/libcrypto.lib" ws2_32 crypt32 secur32 ncrypt bcrypt)
add_executable(ed08_current_normal ../ed08_c3_probe/ed08_c3_probe.c)
target_include_directories(ed08_current_normal PRIVATE "${normal}/include")
target_compile_options(ed08_current_normal PRIVATE /W4 /WX)
target_link_libraries(ed08_current_normal PRIVATE "${normal}/lib/${tid}/papinho_secure_transport.lib" "${normal}/lib/${tid}/libssl.lib" "${normal}/lib/${tid}/libcrypto.lib" ws2_32 crypt32 secur32 ncrypt bcrypt)
foreach(t ed08_r4_probe ed08_current_normal)
 foreach(n libssl-3-x64.dll libcrypto-3-x64.dll)
  add_custom_command(TARGET ${t} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${r4}/runtime/${n}" "$<TARGET_FILE_DIR:${t}>/${n}")
 endforeach()
endforeach()
