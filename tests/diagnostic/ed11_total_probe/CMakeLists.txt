cmake_minimum_required(VERSION 3.16)
project(ed11_total_public_probe C)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Requires documented MSVC19.51 x64 target")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
string(REGEX REPLACE "/W[0-4]" "" CMAKE_C_FLAGS "${CMAKE_C_FLAGS}")
if(NOT PST_SDK_ROOT)
 message(FATAL_ERROR "Exact verified normal SDK root required")
endif()
file(STRINGS "${PST_SDK_ROOT}/manifest.ini" identity)
foreach(line IN ITEMS "api_version=3.0.0" "spi_version=4.0" "package_version=0.6.5" "crt=dynamic-/MD" "source_commit=da7190930700f54ecc92611f5edbcb0932caa5ec")
 list(FIND identity "${line}" found)
 if(found EQUAL -1)
  message(FATAL_ERROR "SDK identity mismatch: ${line}")
 endif()
endforeach()
file(STRINGS "${PST_SDK_ROOT}/manifest.ini" tid REGEX "^target_id=")
string(REPLACE "target_id=" "" tid "${tid}")
set(sdk_lib "${PST_SDK_ROOT}/lib/${tid}")
add_executable(ed11_lifecycle_loop ed11_lifecycle_loop.c)
target_include_directories(ed11_lifecycle_loop PRIVATE "${PST_SDK_ROOT}/include")
target_compile_options(ed11_lifecycle_loop PRIVATE /W4 /WX)
target_link_libraries(ed11_lifecycle_loop PRIVATE "${sdk_lib}/papinho_secure_transport.lib" "${sdk_lib}/libssl.lib" "${sdk_lib}/libcrypto.lib" ws2_32 crypt32 secur32 ncrypt bcrypt psapi)
foreach(name libssl-3-x64.dll libcrypto-3-x64.dll)
 add_custom_command(TARGET ed11_lifecycle_loop POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${PST_SDK_ROOT}/runtime/${tid}/${name}" "$<TARGET_FILE_DIR:ed11_lifecycle_loop>/${name}")
endforeach()

if(NOT ED11_QUEUE_CAP)
 set(ED11_QUEUE_CAP 65536)
endif()
if(NOT ED11_QUEUE_CAP MATCHES "^(8192|16384|32768|65536)$")
 message(FATAL_ERROR "Invalid measurement variant")
endif()
target_compile_definitions(ed11_lifecycle_loop PRIVATE ED11_QUEUE_CAP=${ED11_QUEUE_CAP})

add_executable(ed11_policy_test ed11_policy_test.c)
target_compile_options(ed11_policy_test PRIVATE /W4 /WX)
target_link_libraries(ed11_policy_test PRIVATE psapi)

add_executable(ed11_c1_external_observer ed11_c1_external_observer.c)
target_compile_options(ed11_c1_external_observer PRIVATE /W4 /WX)
target_link_libraries(ed11_c1_external_observer PRIVATE psapi bcrypt)

add_executable(ed11_virtual_map_probe ed11_virtual_map_probe.c)
target_include_directories(ed11_virtual_map_probe PRIVATE "${PST_SDK_ROOT}/include")
target_compile_options(ed11_virtual_map_probe PRIVATE /W4 /WX)
target_link_libraries(ed11_virtual_map_probe PRIVATE "${sdk_lib}/papinho_secure_transport.lib" "${sdk_lib}/libssl.lib" "${sdk_lib}/libcrypto.lib" ws2_32 crypt32 secur32 ncrypt bcrypt psapi)
