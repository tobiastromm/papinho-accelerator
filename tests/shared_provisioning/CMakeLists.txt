cmake_minimum_required(VERSION 3.16)
project(shared_provisioning_structure_offline C)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(shared_structure STATIC "${ROOT}/src/provisioning/shared_structure/structure.c")
target_include_directories(shared_structure PUBLIC "${ROOT}/src/provisioning/shared_structure")
set_target_properties(shared_structure PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
add_executable(structure_differential differential_test.c oracles/manual_package.c oracles/browser_content.c oracles/papinho_papacc_frame.c oracles/browser_password.c oracles/browser_wipe.c)
target_include_directories(structure_differential PRIVATE oracles)
target_link_libraries(structure_differential PRIVATE shared_structure)
target_compile_definitions(structure_differential PRIVATE SP_WITH_ACCELERATOR=1)
set_target_properties(structure_differential PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
add_executable(original_accelerator_regression "${ROOT}/tests/manual_package_test.c" oracles/manual_package.c)
target_include_directories(original_accelerator_regression PRIVATE oracles)
foreach(target shared_structure structure_differential original_accelerator_regression)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
endforeach()
add_test(NAME shared.provisioning.differential COMMAND structure_differential)
add_test(NAME shared.provisioning.original_accelerator COMMAND original_accelerator_regression)
