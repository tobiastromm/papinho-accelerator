cmake_minimum_required(VERSION 3.16)
project(shared_m1_control_offline C)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(shared_m1 m1_test.c "${ROOT}/src/provisioning/shared_structure/m1.c")
target_include_directories(shared_m1 PRIVATE "${ROOT}/src/provisioning/shared_structure")
set_target_properties(shared_m1 PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
 target_compile_options(shared_m1 PRIVATE /W4 /WX)
else()
 target_compile_options(shared_m1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME shared.m1.control COMMAND shared_m1)
