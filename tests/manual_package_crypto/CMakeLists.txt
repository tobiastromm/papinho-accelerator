cmake_minimum_required(VERSION 3.20)
project(papacc_manual_crypto_offline C)
enable_testing()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Only the explicitly qualified modern Windows x64 artifact is selected")
endif()
if(NOT PAPACC_MANUAL_CRYPTO_OPENSSL_358_OFFLINE)
 message(FATAL_ERROR "Explicit offline backend selection required; no fallback")
endif()
if(NOT DEFINED PAPACC_PACKAGE_OPENSSL_ROOT)
 message(FATAL_ERROR "Explicit pinned OpenSSL artifact root required")
endif()
foreach(item "lib/libcrypto.lib|e6047dadbf2b9c85b3bd52598fd6dbac78f9789de9b1c0a837712584ac883509" "runtime/libcrypto-3-x64.dll|09eec573c9adea156ba2073f8cd61720d0aabeb7562d8498b4ecd21b710a3044")
 string(REPLACE "|" ";" pair "${item}")
 list(GET pair 0 name)
 list(GET pair 1 expected)
 file(SHA256 "${PAPACC_PACKAGE_OPENSSL_ROOT}/${name}" actual)
 if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "Pinned crypto artifact mismatch: ${name}")
 endif()
endforeach()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(package_crypto STATIC "${ROOT}/src/provisioning/manual_package.c" "${ROOT}/src/provisioning/manual_package_crypto_openssl.c"
 "${ROOT}/src/provisioning/shared_structure/structure.c" "${ROOT}/src/provisioning/shared_structure/suite.c")
target_include_directories(package_crypto PUBLIC "${ROOT}/src/provisioning" PRIVATE "${PAPACC_PACKAGE_OPENSSL_ROOT}/include")
target_compile_options(package_crypto PRIVATE /W4 /WX)
target_link_libraries(package_crypto PUBLIC "${PAPACC_PACKAGE_OPENSSL_ROOT}/lib/libcrypto.lib")
add_executable(package_crypto_test "${ROOT}/tests/manual_package_crypto_test.c")
target_include_directories(package_crypto_test PRIVATE "${PAPACC_PACKAGE_OPENSSL_ROOT}/include")
target_compile_definitions(package_crypto_test PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(package_crypto_test PRIVATE /W4 /WX)
target_link_libraries(package_crypto_test PRIVATE package_crypto bcrypt)
add_custom_command(TARGET package_crypto_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PAPACC_PACKAGE_OPENSSL_ROOT}/runtime/libcrypto-3-x64.dll" "$<TARGET_FILE_DIR:package_crypto_test>")
add_test(NAME papacc.manual_package.real_crypto COMMAND package_crypto_test "${ROOT}/tests/fixtures/manual-package/upstream-public-test-key.pem" "${ROOT}/tests/fixtures/security/client-a.der.b64" "${ROOT}/tests/fixtures/security/client-a.pk8.b64" "${ROOT}/tests/fixtures/security/client-b.pk8.b64")

# Explicit dependencies for localized /showIncludes hosts.
set_property(SOURCE "${ROOT}/src/provisioning/manual_package_crypto_openssl.c"
 "${ROOT}/src/provisioning/shared_structure/suite.c" APPEND PROPERTY OBJECT_DEPENDS
 "${ROOT}/src/provisioning/shared_structure/suite.h;${ROOT}/src/provisioning/shared_structure/structure.h")
