cmake_minimum_required(VERSION 3.16)
project(PapaccStrictEd25519Qualification C)
add_library(papacc_strict_ed25519 STATIC ../../../src/issuer/strict_ed25519.c)
target_include_directories(papacc_strict_ed25519 PRIVATE ../../../include)
set_target_properties(papacc_strict_ed25519 PROPERTIES C_STANDARD 90 C_EXTENSIONS OFF)
add_executable(strict_ed25519_test ../../../tests/strict_ed25519_test.c)
target_include_directories(strict_ed25519_test PRIVATE ../../../src/issuer)
target_link_libraries(strict_ed25519_test PRIVATE papacc_strict_ed25519)
if(MSVC)
 target_compile_options(papacc_strict_ed25519 PRIVATE /W4 /WX)
 target_compile_options(strict_ed25519_test PRIVATE /W4 /WX)
endif()
enable_testing()
add_test(NAME strict_point_data COMMAND strict_ed25519_test)

add_library(papacc_time_source_key STATIC ../../../src/config/config_root8_time_source_key.c)
target_include_directories(papacc_time_source_key PRIVATE ../../../include)
target_link_libraries(papacc_time_source_key PRIVATE papacc_strict_ed25519)
set_target_properties(papacc_time_source_key PROPERTIES C_STANDARD 90 C_EXTENSIONS OFF)
add_executable(time_source_key_test ../../../tests/config_root8_time_source_key_test.c)
target_include_directories(time_source_key_test PRIVATE ../../../src/config ../../../src/issuer ../../../include)
target_link_libraries(time_source_key_test PRIVATE papacc_time_source_key)
if(MSVC)
 target_compile_options(papacc_time_source_key PRIVATE /W4 /WX)
 target_compile_options(time_source_key_test PRIVATE /W4 /WX)
endif()
add_test(NAME time_source_key_data COMMAND time_source_key_test)
