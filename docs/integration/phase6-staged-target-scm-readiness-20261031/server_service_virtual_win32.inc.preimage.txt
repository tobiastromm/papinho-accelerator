/* Private extension of the SAME SCM owner; no positive authority producer.
 * Immutable executable observation is necessary, not sufficient installation proof. */
static const wchar_t ps_account[]=PAPACC_SCM_ACCOUNT_LITERAL;
static const wchar_t *ps_install_suffix(void)
{
 if(native.install_profile==PS_INSTALL_PRODUCT)return L"\" service host";
#ifndef PAPACC_SCM_REGISTRATION_LAB
 if(native.install_profile==PS_INSTALL_SCQ_CLOSED)return L"\" service host scq-closed";
#endif
 return NULL;
}
/* Same held image and explicit profile at prepare, effect and readback. No
 * interpolated argv, silent diagnostic default or fallback to product mode. */
static int ps_install_command(int capture)
{
 const wchar_t *suffix=ps_install_suffix();size_t n=0,tail;
 if(!suffix)return 0;
 while(n<sizeof(native.image_path)/sizeof(wchar_t)&&native.image_path[n]){
  if(native.image_path[n]==L'"'||native.image_path[n]<L' ')return 0;n++;
 }
 if(n<3U||n==sizeof(native.image_path)/sizeof(wchar_t)||
 !((native.image_path[0]>=L'A'&&native.image_path[0]<=L'Z')||(native.image_path[0]>=L'a'&&native.image_path[0]<=L'z'))||
 native.image_path[1]!=L':'||native.image_path[2]!=L'\\')return 0;
 tail=wcslen(suffix)+1U;
 if(1U+n+tail>sizeof(native.command)/sizeof(wchar_t))return 0;
 if(capture){native.command[0]=L'"';memcpy(native.command+1,native.image_path,n*sizeof(wchar_t));memcpy(native.command+1+n,suffix,tail*sizeof(wchar_t));return 1;}
 return native.command[0]==L'"'&&!memcmp(native.command+1,native.image_path,n*sizeof(wchar_t))&&
 !memcmp(native.command+1+n,suffix,tail*sizeof(wchar_t));
}
static int ps_image_same(const BY_HANDLE_FILE_INFORMATION *a,const BY_HANDLE_FILE_INFORMATION *b)
{
 return a->dwVolumeSerialNumber==b->dwVolumeSerialNumber&&a->nFileIndexHigh==b->nFileIndexHigh&&
 a->nFileIndexLow==b->nFileIndexLow&&a->nFileSizeHigh==b->nFileSizeHigh&&a->nFileSizeLow==b->nFileSizeLow&&
 a->ftLastWriteTime.dwHighDateTime==b->ftLastWriteTime.dwHighDateTime&&a->ftLastWriteTime.dwLowDateTime==b->ftLastWriteTime.dwLowDateTime;
}
static DWORD ps_image_open(void)
{
 DWORD n,read,e=ERROR_INVALID_DATA;BY_HANDLE_FILE_INFORMATION after;PACC_STORAGE_SHA256 hash;
 unsigned char block[1024],digest[32];size_t i;HANDLE f;
 PSRT_CONTEXT("CURRENT_MODULE_FILE");
 if(native.image||native.image_ready)return PSRT_RETURN(ERROR_INVALID_STATE,"original_branch");
 n=PSRT_COUNT("GetModuleFileNameW",GetModuleFileNameW(NULL,native.image_path,1024));
 if(!n)return PSRT_RETURN(ps_last_error(),"native_api_failure");if(n>=1024)return PSRT_RETURN(ERROR_INSUFFICIENT_BUFFER,"original_branch");
 if(!((native.image_path[0]>=L'A'&&native.image_path[0]<=L'Z')||(native.image_path[0]>=L'a'&&native.image_path[0]<=L'z'))||
 native.image_path[1]!=L':'||native.image_path[2]!=L'\\')return PSRT_RETURN(ERROR_INVALID_NAME,"original_branch");
 for(i=0;i<n;i++)if(native.image_path[i]==L'"'||native.image_path[i]<L' ')return PSRT_RETURN(ERROR_INVALID_NAME,"original_branch");
 f=PSRT_HANDLE("CreateFileW",(GENERIC_READ),CreateFileW(native.image_path,GENERIC_READ,FILE_SHARE_READ,NULL,OPEN_EXISTING,FILE_FLAG_OPEN_REPARSE_POINT,NULL));
 if(f==INVALID_HANDLE_VALUE)return PSRT_RETURN(ps_last_error(),"native_api_failure");native.image=f;
 if(!PSRT_BOOL("GetFileInformationByHandle",(FILE_READ_ATTRIBUTES),GetFileInformationByHandle(f,&native.image_id)))return PSRT_RETURN(ps_last_error(),"native_api_failure");
 if(native.image_id.dwFileAttributes&(FILE_ATTRIBUTE_REPARSE_POINT|FILE_ATTRIBUTE_DIRECTORY))return PSRT_RETURN(ERROR_ACCESS_DENIED,"image.not_reparse_or_directory");
 memset(&hash,0,sizeof(hash));if(pacc_storage_sha_init(&hash)!=PACC_HASH_OK)return PSRT_RETURN(ERROR_INVALID_DATA,"original_branch");
 for(;;){
  if(papacc_server_console_win32_stop_requested()) {e=ERROR_OPERATION_ABORTED;goto done;}
  if(!PSRT_BOOL("ReadFile",(FILE_READ_DATA),ReadFile(f,block,sizeof(block),&read,NULL))){e=ps_last_error();goto done;}
  if(!read)break;
  if(pacc_storage_sha_update(&hash,block,read,NULL,NULL)!=PACC_HASH_OK)goto done;
 }
 if(pacc_storage_sha_final(&hash,digest,sizeof(digest),NULL,NULL)!=PACC_HASH_OK)goto done;
 if(!PSRT_BOOL("GetFileInformationByHandle",(FILE_READ_ATTRIBUTES),GetFileInformationByHandle(f,&after))){e=ps_last_error();goto done;}
 if(!ps_image_same(&native.image_id,&after)||papacc_server_console_win32_stop_requested())goto done;
 if(!ps_install_command(1))goto done;
 memcpy(native.image_sha256,digest,32);native.image_ready=1U;e=ERROR_SUCCESS;
 done:memset(&hash,0,sizeof(hash));memset(block,0,sizeof(block));memset(digest,0,sizeof(digest));return PSRT_RETURN(e,"module_image_read_result");
}
/* Never dereference returned string pointers before validating buffer range,
 * alignment and terminator inside the actually returned byte count. */
static int ps_config_string(const wchar_t *s,DWORD used,const wchar_t *expected,int insensitive)
{
 ULONG_PTR begin=(ULONG_PTR)native.query.bytes,end,p=(ULONG_PTR)s;size_t count,i;
 if(used>sizeof(native.query.bytes)||used<sizeof(QUERY_SERVICE_CONFIGW))return 0;
 end=begin+used;
 if(!s||p<begin+sizeof(QUERY_SERVICE_CONFIGW)||p>=end||(p-begin)%sizeof(wchar_t))return 0;
 count=(size_t)((end-p)/sizeof(wchar_t));for(i=0;i<count;i++)if(!s[i])break;
 if(i==count||i!=wcslen(expected))return 0;
 return insensitive?_wcsicmp(s,expected)==0:wcscmp(s,expected)==0;
}
static DWORD ps_virtual_readback_start(DWORD expected_start)
{
 QUERY_SERVICE_CONFIGW *q=&native.query.aligned;SERVICE_SID_INFO info;DWORD used=0,n=0,sidn,domainn;
 SID_NAME_USE use;BYTE authority[6]={0,0,0,0,0,5};PSID sid=native.service_sid;
 if(!ps_install_command(0))return ERROR_INVALID_DATA;
 PSRT_CONTEXT("SCM_SERVICE_READBACK");
 memset(&native.query,0,sizeof(native.query));
 if(!PSRT_BOOL("QueryServiceConfigW",(SERVICE_QUERY_CONFIG),QueryServiceConfigW(native.service,q,sizeof(native.query.bytes),&used)))return PSRT_RETURN(ps_last_error(),"native_api_failure");
 if(used>sizeof(native.query.bytes)||used<sizeof(*q)||q->dwServiceType!=SERVICE_WIN32_OWN_PROCESS||
 q->dwStartType!=expected_start||!ps_config_string(q->lpServiceStartName,used,ps_account,1)||
 !ps_config_string(q->lpBinaryPathName,used,native.command,0)||
 (q->lpDependencies&&!ps_config_string(q->lpDependencies,used,L"",0)))return PSRT_RETURN(ERROR_INVALID_DATA,"original_branch");
 memset(&info,0,sizeof(info));
 if(!PSRT_BOOL("QueryServiceConfig2W",(SERVICE_QUERY_CONFIG),QueryServiceConfig2W(native.service,SERVICE_CONFIG_SERVICE_SID_INFO,(LPBYTE)&info,sizeof(info),&n)))return PSRT_RETURN(ps_last_error(),"native_api_failure");
 if(n!=sizeof(info)||info.dwServiceSidType!=SERVICE_SID_TYPE_UNRESTRICTED)return PSRT_RETURN(ERROR_INVALID_DATA,"original_branch");
 sidn=sizeof(native.service_sid);domainn=sizeof(native.sid_domain)/sizeof(wchar_t);
 if(!PSRT_BOOL("LookupAccountNameW",(0),LookupAccountNameW(NULL,ps_account,sid,&sidn,native.sid_domain,&domainn,&use)))return PSRT_RETURN(ps_last_error(),"native_api_failure");
 if(sidn>sizeof(native.service_sid)||sidn<8U||!IsValidSid(sid)||GetLengthSid(sid)>sidn||
 *GetSidSubAuthorityCount(sid)!=6||memcmp(GetSidIdentifierAuthority(sid),authority,6)||*GetSidSubAuthority(sid,0)!=80)return PSRT_RETURN(ERROR_INVALID_SID,"original_branch");
 native.sid_size=GetLengthSid(sid);native.sid_use=use;native.readback=1U;return ERROR_SUCCESS;
}
static DWORD ps_virtual_readback(void){return ps_virtual_readback_start(PAPACC_SCM_START_MODE);}
static DWORD ps_install_virtual(void)
{
 SERVICE_SID_INFO info;DWORD e;BY_HANDLE_FILE_INFORMATION current;
 /* Caller must first prove configuration/target/authorization via existing
  * gate. This private primitive is independently tested with modeled APIs. */
 if(!native.manager||native.service||!native.image||!native.image_ready||native.install_effect)return PSRT_RETURN(ERROR_INVALID_STATE,"original_branch");
 if(!ps_install_command(0))return ERROR_INVALID_DATA;
 if(papacc_server_console_win32_stop_requested())return PSRT_RETURN(ERROR_OPERATION_ABORTED,"original_branch");
 if(!GetFileInformationByHandle(native.image,&current))return ps_last_error();
 if(!ps_image_same(&native.image_id,&current))return PSRT_RETURN(ERROR_INVALID_DATA,"original_branch");
 if(native.install_profile==PS_INSTALL_SCQ_CLOSED){
  if(!native.registration_active)return ERROR_ACCESS_DENIED;
  e=papacc_server_launcher_scm_registration_take(&native.registration_view);if(e)return e;
 }
 native.install_effect=1U; /* REQUESTED: an uncertain Create is never retried. */
 native.service=CreateServiceW(native.manager,ps_name,ps_name,SERVICE_QUERY_CONFIG|SERVICE_CHANGE_CONFIG|SERVICE_QUERY_STATUS,
 SERVICE_WIN32_OWN_PROCESS,PAPACC_SCM_START_MODE,SERVICE_ERROR_NORMAL,native.command,NULL,NULL,NULL,ps_account,NULL);
 if(!native.service){e=ps_last_error();native.install_disposition=e==ERROR_SERVICE_EXISTS||e==ERROR_DUPLICATE_SERVICE_NAME||e==ERROR_SERVICE_MARKED_FOR_DELETE?1U:2U;return PSRT_RETURN(e,"propagate_original");}
 native.install_disposition=3U; /* Created, not configured/finalized runtime. */
 if(papacc_server_console_win32_stop_requested())return PSRT_RETURN(ERROR_OPERATION_ABORTED,"original_branch");
 if(native.install_profile==PS_INSTALL_SCQ_CLOSED){e=papacc_server_launcher_scm_registration_current(&native.registration_view);if(e)return e;}
 info.dwServiceSidType=SERVICE_SID_TYPE_UNRESTRICTED;
 if(!ChangeServiceConfig2W(native.service,SERVICE_CONFIG_SERVICE_SID_INFO,&info))return ps_last_error();
 e=ps_virtual_readback();if(e)return PSRT_RETURN(e,"propagate_original");
 if(papacc_server_console_win32_stop_requested())return PSRT_RETURN(ERROR_OPERATION_ABORTED,"original_branch");
 if(native.install_profile==PS_INSTALL_SCQ_CLOSED){e=papacc_server_launcher_scm_registration_current(&native.registration_view);if(e)return e;}
 native.install_disposition=4U; /* Configuration confirmed, NOT token/ACL/readiness. */
 return ERROR_SUCCESS;
}
